module TB_All_Gates();
reg A,B,ctrl;
wire [9:1] Y;

All_Gates UUT(.A(A),.B(B),.ctrl(ctrl),.Y(Y));
initial
begin
        A=0; B=0; ctrl=0; #10;
        A=0; B=1; ctrl=0; #10;
        A=1; B=0; ctrl=0; #10;
        A=1; B=1; ctrl=0; #10;

        A=0; B=0; ctrl=1; #10;
        A=0; B=1; ctrl=1; #10;
        A=1; B=0; ctrl=1; #10;
        A=1; B=1; ctrl=1; #10;
end
endmodule
